\section{理论模型}
\subsection{线性频域方程}

% 版式三：公式主导——公式为主体，说明为次级信息。
\begin{frame}[t]{线性频域运动方程}
  \ResearchLead{运动响应由质量、辐射阻尼、恢复力与波浪激励共同确定}

  \vspace{.25em}
  \begin{equation*}
  \left[
    -\omega^2\bigl(\mathbf{M}+\mathbf{A}(\omega)\bigr)
    +\mathrm{i}\omega\mathbf{B}(\omega)
    +\mathbf{C}
  \right]\hat{\boldsymbol{\xi}}(\omega)
  =\hat{\mathbf{F}}_{\mathrm{exc}}(\omega).
  \end{equation*}

  \vspace{.35em}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.47\textwidth}
      \ResearchHeading{惯性与阻尼}
      $\mathbf{M}$：结构质量及转动惯量矩阵\par\smallskip
      $\mathbf{A}(\omega)$：附加质量矩阵\par\smallskip
      $\mathbf{B}(\omega)$：辐射阻尼矩阵
    \end{column}
    \begin{column}{.47\textwidth}
      \ResearchHeading{恢复力与响应}
      $\mathbf{C}$：线性恢复力矩阵\par\smallskip
      $\hat{\mathbf{F}}_{\mathrm{exc}}$：波浪激励幅值\par\smallskip
      $\hat{\boldsymbol{\xi}}$：六自由度复响应幅值
    \end{column}
  \end{columns}
\end{frame}

\subsection{响应指标}
\begin{frame}[t]{响应幅值算子（RAO）}
  \ResearchLead{以入射波幅值归一化，比较不同频率下的运动响应}

  \begin{equation*}
    \operatorname{RAO}_{j}(\omega)
       = \frac{\left|\hat{\xi}_{j}(\omega)\right|}{a},
       \qquad j=1,\ldots,6 .
  \end{equation*}

  \vspace{.6em}
  \begin{ResearchNote}
    对平动自由度，RAO 可表示为无量纲位移响应比；
    对转动自由度，应根据角位移及选定的归一化方式明确单位。
    这里以规则波幅值 $a$ 为示例说明。
  \end{ResearchNote}
\end{frame}
