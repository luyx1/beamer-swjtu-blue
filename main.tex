\begin{frame}{模板说明}
  \begin{itemize}
    \item 本模板面向西南交通大学科研汇报场景。
    \item 主题文件为 \texttt{beamerthemeswjtu.sty} 及其配套的颜色、内部和外部主题文件。
    \item 校徽、校楼和验证图片位于 \texttt{resources/swjtu\_*.{pdf,png}}。
    \item 默认使用 XeLaTeX 编译；当前示例不启用参考文献。
  \end{itemize}
\end{frame}

\begin{frame}{建议的文件组织}
  \begin{block}{主文件}
    在 \texttt{main.tex} 中填写标题、作者、单位、日期和正文内容。
  \end{block}
  \begin{block}{资源文件}
    将图片放入 \texttt{resources/}、\texttt{figure/} 或 \texttt{figures/}，主题会自动搜索这些目录。
  \end{block}
  \begin{alertblock}{编译提示}
    如果启用参考文献，请将主题选项改为 \texttt{BIBMode=biber}，并使用 XeLaTeX + Biber 编译。
  \end{alertblock}
\end{frame}
